HTML & JavaScript Example

Minimal setup to embed the Top Movers chart in your own page. Provide three buckets of data per user (w0 = current, w1 = previous, w2 = two periods ago) and the scoring engine handles the rest.

Include the script, point it at two container elements, and call three functions. 

<!-- 1. Styles + script -->
<link rel="stylesheet" href="components/custom/dumbbell-chart/styles.css">
<script src="components/custom/dumbbell-chart/dumbbell-chart.js"></script>

<!-- 2. Chart containers -->
<div id="chartIncrease" class="dumbbell-panel"></div>
<div id="chartDecrease" class="dumbbell-panel"></div>
const DC = window.DumbbellChart;

// Supply your own data: each record needs { name, w0, w1, w2 }
// w0 = current period  ·  w1 = previous period  ·  w2 = two periods ago
const data = DC.generateSampleData(); // ← replace with real data

const cfg = {
  minCalls:    1000,   // filter out users below this threshold
  smooth:       100,   // additive smoothing for log-ratios
  winsorLow:      5,   // clip z-score inputs below this percentile
  winsorHigh:    95,   // clip z-score inputs above this percentile
  weights: { absolute: 0.70, ratio: 0.25, trend: 0.05 },
  reliability: { enabled: true, baselineRef: 10000 },
  topN: 10
};

const scored = DC.computeScores(data, cfg);
const movers = DC.getTopMovers(scored, cfg.topN);

DC.createChart(document.getElementById('chartIncrease'), {
  rows:      movers.increases,
  direction: 'increase',
  logScale:  false,
  xAxisMin:  cfg.minCalls,
  title:     'Top Increases'
});

DC.createChart(document.getElementById('chartDecrease'), {
  rows:      movers.decreases,
  direction: 'decrease',
  logScale:  false,
  xAxisMin:  cfg.minCalls,
  title:     'Top Decreases'
});
