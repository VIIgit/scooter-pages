aggregates(data, mode)

Helper function from index.html: plain data is transformed into grouped high/low/average structures.

Aggregated by month

Simple date/value records become a high-low compatible structure

<div class="barchart-item">
  <div class="chart-container barchart-host" id="chart-aggregates"></div>
  <div class="barchart-output" id="aggregates-output"></div>
</div>
var plainData = [
  { date: '2025-01-01', value: 42 },
  { date: '2025-01-02', value: 58 },
  { date: '2025-01-15', value: 35 },
  { date: '2025-02-03', value: 71 },
  { date: '2025-02-20', value: 49 },
  { date: '2025-03-10', value: 62 }
];

var aggregatedByMonth = Barchart.aggregates(plainData, 'byMonth');

Barchart.createChart({
  container: '#chart-aggregates',
  chartType: 'byMonth',
  visibleWidth: 640,
  chartHeight: 220,
  timeSeries: [{
    data: plainData,
    renderType: 'high-low',
    title: 'Aggregated from Plain Data',
    yAxisLabel: 'Value'
  }]
});

var outputEl = document.querySelector('#aggregates-output');
if (outputEl) {
  outputEl.textContent = JSON.stringify(aggregatedByMonth, null, 2);
}

console.log('[demo-barchart] Aggregates helper result', aggregatedByMonth);
