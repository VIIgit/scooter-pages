A full-featured code editor with syntax highlighting, minimap, IntelliSense and more. Use the controls below to change language and theme on the fly.

/** * Example: Fibonacci sequence generator * Try editing this code — the editor supports full IntelliSense! */ function* fibonacci(limit = Infinity) { let a = 0, b = 1; let count = 0; while (count < limit) { yield a; [a, b] = [b, a + b]; count++; } } // Generate first 20 Fibonacci numbers const fibs = [...fibonacci(20)]; console.log('Fibonacci:', fibs); // Calculate the golden ratio approximation const ratio = fibs[fibs.length - 1] / fibs[fibs.length - 2]; console.log('Golden ratio ≈', ratio.toFixed(10));

interface ScooterComponent { name: string; tier: 'css' | 'light' | 'medium' | 'complex'; init: (el: HTMLElement) => void; } type ComponentRegistry = Map<string, ScooterComponent>; const registry: ComponentRegistry = new Map(); function register<T extends ScooterComponent>(component: T): void { if (registry.has(component.name)) { console.warn(`Component "${component.name}" already registered`); return; } registry.set(component.name, component); } function initAll(root: HTMLElement = document.body): void { registry.forEach((comp, name) => { const elements = root.querySelectorAll<HTMLElement>( `[data-sc="${name}"]` ); elements.forEach(el => comp.init(el)); }); } export { register, initAll, type ScooterComponent };

Diff Editor

Side-by-side diff viewer that highlights changes between an original and modified version. Provide content via data-slot="original" and data-slot="modified" child elements. 

/**
 * UserService — original version
 */
class UserService {
  constructor(db) {
    this.db = db;
  }

  async getUser(id) {
    const user = await this.db.query('SELECT * FROM users WHERE id = ?', [id]);
    return user;
  }

  async createUser(name, email) {
    const result = await this.db.query(
      'INSERT INTO users (name, email) VALUES (?, ?)',
      [name, email]
    );
    return { id: result.insertId, name, email };
  }

  async deleteUser(id) {
    await this.db.query('DELETE FROM users WHERE id = ?', [id]);
  }
}
/**
 * UserService — updated version
 * Added validation, soft-delete, and caching.
 */
class UserService {
  #cache = new Map();

  constructor(db, logger) {
    this.db = db;
    this.logger = logger;
  }

  async getUser(id) {
    if (this.#cache.has(id)) {
      return this.#cache.get(id);
    }

    const user = await this.db.query('SELECT * FROM users WHERE id = ? AND deleted_at IS NULL', [id]);
    if (user) this.#cache.set(id, user);
    return user;
  }

  async createUser(name, email) {
    if (!name || !email) {
      throw new Error('Name and email are required');
    }
    if (!email.includes('@')) {
      throw new Error('Invalid email address');
    }

    const result = await this.db.query(
      'INSERT INTO users (name, email, created_at) VALUES (?, ?, NOW())',
      [name, email]
    );
    const user = { id: result.insertId, name, email };
    this.#cache.set(user.id, user);
    this.logger.info('User created', { id: user.id });
    return user;
  }

  async deleteUser(id) {
    await this.db.query('UPDATE users SET deleted_at = NOW() WHERE id = ?', [id]);
    this.#cache.delete(id);
    this.logger.info('User soft-deleted', { id });
  }
}

Component Features

Overview of what the Monaco Editor wrapper provides.

Singleton CDN Loader

Monaco is loaded exactly once via a shared Promise — no race conditions even with multiple editors on the same page.

Standard + Diff Modes

Switch between a standard code editor and a side-by-side diff viewer with a single data-mode attribute.

Declarative Config

Language, theme, minimap, word-wrap, and more — all controlled via data attributes. No JavaScript setup needed.

Full API Access

Access the underlying Monaco instance via el._monacoEditor.getEditor() for advanced customization.
